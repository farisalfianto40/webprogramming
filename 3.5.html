1. style.css (Stylesheet Eksternal)

style.css 
/* --- Reset & Base Styles --- */
* {
    margin: 0;
    padding: 0;
    box-sizing: border-box;
}

body {
    font-family: 'Segoe UI', Tahoma, Geneva, Verdana, sans-serif;
    background-color: #f1f5f9;
    color: #334155;
}

.container {
    max-width: 1100px;
    margin: 0 auto;
    padding: 1.5rem;
}

/* --- Header & Navigation --- */
.navbar {
    background-color: #1e293b;
    padding: 1rem 2rem;
    display: flex;
    justify-content: space-between;
    align-items: center;
    border-radius: 0.5rem;
    margin-bottom: 1.5rem;
}

.brand {
    color: #ffffff;
    font-weight: bold;
    font-size: 1.2rem;
}

.nav-links {
    display: flex;
    gap: 1rem;
    list-style: none;
}

.nav-links a {
    color: #cbd5e1;
    text-decoration: none;
    font-weight: 500;
    padding: 0.4rem 0.8rem;
    border-radius: 0.25rem;
    transition: all 0.3s ease;
}

.nav-links a:hover, .nav-links a.active {
    color: #ffffff;
    background-color: #0284c7;
}

/* --- Cards & Content Layout --- */
.card {
    background-color: #ffffff;
    border-radius: 0.5rem;
    box-shadow: 0 4px 6px -1px rgba(0, 0, 0, 0.1);
    padding: 1.5rem;
    margin-bottom: 1.5rem;
    border: 1px solid #e2e8f0;
}

.hero {
    background-color: #0f172a;
    color: #ffffff;
    border-radius: 0.75rem;
    padding: 2.5rem 2rem;
}

.grid-3 {
    display: grid;
    grid-template-columns: repeat(auto-fit, minmax(250px, 1fr));
    gap: 1rem;
    margin-top: 1rem;
}

/* --- Table Component --- */
.table-custom {
    width: 100%;
    border-collapse: collapse;
    margin-top: 1rem;
}

.table-custom th, .table-custom td {
    padding: 0.75rem 1rem;
    border: 1px solid #e2e8f0;
    text-align: left;
}

.table-custom th {
    background-color: #f8fafc;
    color: #0f172a;
}

.text-center {
    text-align: center;
}

.img-cover {
    width: 48px;
    height: 64px;
    object-fit: cover;
    border-radius: 0.25rem;
    border: 1px solid #cbd5e1;
}

/* --- Form Styling --- */
.form-group {
    margin-bottom: 1rem;
}

.form-group label {
    display: block;
    margin-bottom: 0.25rem;
    font-weight: 500;
}

.form-control {
    width: 100%;
    padding: 0.6rem 0.8rem;
    border: 1px solid #cbd5e1;
    border-radius: 0.375rem;
    outline: none;
}

.btn-primary {
    background-color: #0284c7;
    color: white;
    padding: 0.6rem 1.2rem;
    border: none;
    border-radius: 0.375rem;
    cursor: pointer;
    font-weight: 600;
}

.btn-primary:hover {
    background-color: #0369a1;
}
